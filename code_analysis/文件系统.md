## super block

超级块中描述了设备中文件系统的布局，每个存储设备都有一个超级块。

```c++
struct superblock {
  uint magic;        // Must be FSMAGIC
  uint size;         // Size of file system image (blocks)
  uint nblocks;      // Number of data blocks
  uint ninodes;      // Number of inodes.
  uint nlog;         // Number of log blocks
  uint logstart;     // Block number of first log block
  uint inodestart;   // Block number of first inode block
  uint bmapstart;    // Block number of first free map block
};
```

## inode

### 含义

存储文件元数据的索引节点。

### 在磁盘中的结构

在磁盘中所有inode放在一起顺序排列，每个inode都有一个自己的编号:

```c++
struct dinode {
  short type;           // File type
  short major;          // Major device number (T_DEVICE only)
  short minor;          // Minor device number (T_DEVICE only)
  short nlink;          // Number of links to inode in file system
  uint size;            // Size of file (bytes)
  uint addrs[NDIRECT+1];   // Data block addresses
};
```

### 在内存中的结构

下面的inode结构体表示磁盘中inode在内存中的结构，保存了文件的类型、inode编号、引用个数、文件对应的磁盘block的地址addrs等，注意没有文件名。

```c++
struct inode {
  uint dev;           // Device number
  uint inum;          // Inode number
  int ref;            // Reference count
  struct sleeplock lock; // protects everything below here
  int valid;          // inode has been read from disk?

  short type;         // copy of disk inode
  short major;
  short minor;
  short nlink;
  uint size;
  uint addrs[NDIRECT+1];
};
```

内存中载入的inode放在一个数组中：

```c++
struct {
  struct spinlock lock;
  struct inode inode[NINODE];
} itable;
```

### 使用方式

```c++
ip = iget(dev, inum)
ilock(ip)
// ... examine and modify ip->xxx ...
iunlock(ip)
iput(ip)
```

### 创建inode

1. 在磁盘中遍历找到一个空闲的inode位置
2. 在这个位置上进行初始化
3. 计入log
4. 返回新创建的inode

```c++
struct inode* ialloc(uint dev, short type) {
  for(int inum = 1; inum < sb.ninodes; inum++) {
    struct buf *bp = bread(dev, IBLOCK(inum, sb));
    struct dinode *dip = (struct dinode*)bp->data + inum % IPB;
    if(dip->type == 0) {  // a free inode
      memset(dip, 0, sizeof(*dip));
      dip->type = type;
      log_write(bp);   // mark it allocated on the disk
      brelse(bp);
      return iget(dev, inum);
    }
    brelse(bp);
  }
  printf("ialloc: no inodes\n");
  return 0;
}
```

### 根据路径查找inode

1. 首先查看是绝对还是相对路径：
   1. 绝对路径：通过iget获取根目录对应的inode
   2. 相对路径：通过idup获取当前进程的工作路径
2. 解析path，取得第一个子目录，如/a/b/c，取得a，在inode中查找这个在子目录
3. 找到子目录对应的inode后，继续递归遍历，直到找到文件

```c++
static struct inode* namex(char *path, int nameiparent, char *name) {
  struct inode *ip, *next;

  if(*path == '/')
    ip = iget(ROOTDEV, ROOTINO);
  else
    ip = idup(myproc()->cwd);

  while((path = skipelem(path, name)) != 0){
    ilock(ip);
    if(ip->type != T_DIR){
      iunlockput(ip);
      return 0;
    }
    if(nameiparent && *path == '\0'){
      // Stop one level early.
      iunlock(ip);
      return ip;
    }
    if((next = dirlookup(ip, name, 0)) == 0){
      iunlockput(ip);
      return 0;
    }
    iunlockput(ip);
    ip = next;
  }
  if(nameiparent){
    iput(ip);
    return 0;
  }
  return ip;
}
```

#### 根据文件名查找文件

inode中没有存放文件名，那文件名存放在哪里了？

文件名存放在父目录中，dirlookup是在文件的父目录ip中查找name：

```c++
struct inode* dirlookup(struct inode *dp, char *name, uint *poff) {
  uint off, inum;
  struct dirent de;

  if(dp->type != T_DIR)
    panic("dirlookup not DIR");

  for(off = 0; off < dp->size; off += sizeof(de)){
    if(readi(dp, 0, (uint64)&de, off, sizeof(de)) != sizeof(de))
      panic("dirlookup read");
    if(de.inum == 0)
      continue;
    if(namecmp(name, de.name) == 0){
      // entry matches path element
      if(poff)
        *poff = off;
      inum = de.inum;
      return iget(dp->dev, inum);
    }
  }

  return 0;
}
```
